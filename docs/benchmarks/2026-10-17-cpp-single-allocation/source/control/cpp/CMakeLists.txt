cmake_minimum_required(VERSION 3.20)
project(war_modern_cpp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WAR_UWS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../artifacts/modern-uwebsockets" CACHE PATH "Pinned patched uWebSockets source; run cpp/setup.sh first")
if(NOT EXISTS "${WAR_UWS_SOURCE}/src/App.h")
  message(FATAL_ERROR "Run cpp/setup.sh before configuring")
endif()
execute_process(COMMAND git -C "${WAR_UWS_SOURCE}" apply --reverse --check "${CMAKE_CURRENT_SOURCE_DIR}/patches/war-transport.patch" RESULT_VARIABLE patch_result)
if(NOT patch_result EQUAL 0)
  message(FATAL_ERROR "Pinned transport patch is missing; run cpp/setup.sh")
endif()
file(GLOB usockets_sources CONFIGURE_DEPENDS "${WAR_UWS_SOURCE}/uSockets/src/*.c" "${WAR_UWS_SOURCE}/uSockets/src/eventing/*.c" "${WAR_UWS_SOURCE}/uSockets/src/crypto/*.c" "${WAR_UWS_SOURCE}/uSockets/src/io_uring/*.c")
add_library(war_usockets STATIC ${usockets_sources})
target_compile_definitions(war_usockets PUBLIC LIBUS_NO_SSL)
target_include_directories(war_usockets SYSTEM PUBLIC "${WAR_UWS_SOURCE}/uSockets/src")
add_library(war-relay STATIC relay.cpp)
target_include_directories(war-relay PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(war-relay SYSTEM PRIVATE "${WAR_UWS_SOURCE}/src")
target_compile_definitions(war-relay PRIVATE UWS_NO_ZLIB LIBUS_NO_SSL)
target_link_libraries(war-relay PUBLIC war_usockets pthread)
add_executable(war-relay-cpp main.cpp)
target_link_libraries(war-relay-cpp PRIVATE war-relay)
add_executable(war-cbor-check tests/cbor.cpp)
enable_testing()
add_test(NAME cbor-boundaries COMMAND war-cbor-check)
add_executable(war-embed-check tests/embed.cpp)
target_link_libraries(war-embed-check PRIVATE war-relay)
add_test(NAME embedded-stop-token COMMAND war-embed-check)
